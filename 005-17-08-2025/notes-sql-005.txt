4- SELECT : Used to retrieve data from a table(S).
    ....

    - logic operations : AND, OR, NOT

        >>> SELECT `CompanyName`,`Country`,`City` FROM `customers` WHERE `Country`='germany' AND `City`='berlin'

        >>> SELECT `ProductName`,`UnitPrice`,`UnitsInStock` FROM `products` WHERE NOT(`UnitPrice` > 100 AND `UnitsInStock` = 0);

        >>> SELECT `CompanyName`,`Country` FROM `suppliers`
            WHERE `CompanyName` LIKE 's%' AND NOT(`Country` = 'usa' OR `Country` = 'uk')

        >>> SELECT `CompanyName`,`Country` FROM `suppliers` WHERE `CompanyName` LIKE 's%' AND `Country` NOT IN('usa','uk');



    - calculated Fields

        >>> SELECT `ProductName`,`UnitPrice`,`UnitsInStock`, (`UnitPrice` * `UnitsInStock`) AS productTotal FROM `products` WHERE 1;

        >>> SELECT CONCAT(`TitleOfCourtesy`, ' ', `FirstName`, ' ', `LastName`, ' - ', `Title`) AS empName FROM `employees` WHERE 1;



    - aggregate functions : MIN, MAX, COUNT, AVG, SUM
                            - GROUP BY, HAVING

    >>> SELECT `ProductName`, MAX(`UnitPrice`) AS maxPrice FROM `products` WHERE 1;

    >>> SELECT `Country`, COUNT(`SupplierID`) AS SupppliersCount FROM `suppliers` GROUP BY `Country`;

    >>> SELECT `Country`, COUNT(`SupplierID`) AS SupppliersCount FROM `suppliers` GROUP BY `Country` HAVING `Country` = 'usa';

    >>> SELECT `Country`, COUNT(`SupplierID`) AS SupppliersCount FROM `suppliers` WHERE `Country` = 'usa' GROUP BY `Country`;

    >>> SELECT `Country`, COUNT(`SupplierID`) AS SupppliersCount FROM `suppliers` GROUP BY `Country` HAVING `SupppliersCount` >= 3;




    - relationships with SELECT
        >>> SELECT `categories`.`CategoryName`, `products`.`ProductName` FROM `categories`, `products` WHERE `categories`.`CategoryID` = `products`.`CategoryID` ORDER BY `categories`.`CategoryID`;

        >>> SELECT `categories`.`CategoryName`, `products`.`ProductName`, `order details`.`UnitPrice`, `order details`.`Quantity` FROM `categories`, `products`,`order details` WHERE `categories`.`CategoryID` = `products`.`CategoryID` AND `products`.`ProductID` = `order details`.`ProductID` AND `order details`.`OrderID` = 10248;

        >>> SELECT `order details`.`OrderID`, `customers`.`CompanyName`, `orders`.`OrderDate`, SUM(`order details`.`UnitPrice` * `order details`.`Quantity`) AS orderTotal FROM `order details`, `orders`, `customers` WHERE `customers`.`CustomerID` = `orders`.`CustomerID` AND `orders`.`OrderID` = `order details`.`OrderID` AND `order details`.`OrderID` = 10248 GROUP BY `order details`.`OrderID`;

        >>> SELECT `order details`.`OrderID`, `customers`.`CompanyName`, `orders`.`OrderDate`, SUM(`order details`.`UnitPrice` * `order details`.`Quantity`) AS orderTotal FROM `order details`, `orders`, `customers` WHERE `customers`.`CustomerID` = `orders`.`CustomerID` AND `orders`.`OrderID` = `order details`.`OrderID` GROUP BY `order details`.`OrderID`;
