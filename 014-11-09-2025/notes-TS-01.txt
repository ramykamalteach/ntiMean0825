
Type Script

TypeScript is a programming language developed by Microsoft.
It is essentially JavaScript with extra features, mainly:

1- Superset of JavaScript
    - Any valid JavaScript code is also valid TypeScript.
    - This means you can start with JavaScript and gradually add TypeScript features.

2- Static Typing
    - Unlike JavaScript, which is dynamically typed, TypeScript allows you to add types (like string, number, boolean, object, array, etc.) to your variables, functions, and classes.

3- Error Checking at Compile Time
    - TypeScript catches many errors before you run the code, which reduces runtime bugs.

4- Modern JavaScript Features + More
    - Supports ES6+ features (classes, modules, async/await, etc.).
    - Adds features not in JavaScript yet (interfaces, enums, generics, etc.).

5- Compiles to JavaScript
    - Browsers don’t understand TypeScript directly.
    - TypeScript must be compiled into plain JavaScript using the TypeScript compiler (tsc).

-- Optional Parameters

function greetUser(name: string, age?: number): string {
  return age ? `Hello, ${name}. You are ${age}.` : `Hello, ${name}`;
}

console.log(greetUser("Samy"));      // Hello, Ramy
console.log(greetUser("Samy", 30));  // Hello, Ramy. You are 30.


-- Default Parameters

function multiply(a: number, b: number = 2): number {
  return a * b;
}

console.log(multiply(5));    // 10
console.log(multiply(5, 3)); // 15


-- Arrow Functions

const square = (x: number): number => {
  return x * x;
};

console.log(square(4)); // 16


-- Function Type

let operation: (a: number, b: number) => number;

operation = (x, y) => x + y;
console.log(operation(3, 4)); // 7

or

function formatValue(value: string | number, prefix: string = "Value:", suffix?: string): string {
  let result = `${prefix} ${value}`;
  if (suffix) {
    result += ` ${suffix}`;
  }
  return result;
}

console.log(formatValue(100));                    // "Value: 100"
console.log(formatValue("Hello", "Message"));     // "Message Hello"
console.log(formatValue(42, "Number", "✔"));      // "Number 42 ✔"

or

type Operation = (a: number, b: number | string) => string;

const combine: Operation = (a, b) => {
  if (typeof b === "number") {
    return `Sum: ${a + b}`;
  }
  return `Concat: ${a}${b}`;
};

console.log(combine(5, 10));   // Sum: 15
console.log(combine(5, "X"));  // Concat: 5X
