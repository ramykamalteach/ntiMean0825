Build MVC

1- folders architecture :
    - root
        - controllers
        - models
        - views
            - partials
            - pages
        - route
        - public

2- folders/files architecture :
    - root
        - controllers
            - staffController.js
        - models
            - staffModel.js
        - views
            - partials
                - siteHeader.ejs
                - siteFooter.ejs
                - emp.ejs
            - pages
                - home.ejs
                - staff.ejs
        - route
            - web.js
        - public
            - siteStyles.css
        
        - index.js
        - .env


3- install packages
    > npm init -y
    > npm i nodemon
    > npm i express@5
    > npm i path
    > npm i dotenv
    > npm i ejs -g


--------------------

index.js code :

const express = require('express');
const app = express();
const path = require('path');
require('dotenv').config();

/* ------------- public folder ---------------- */
app.use(express.static(__dirname + '/public'));
/* ------------- view engine ------------------ */
app.set('view engine', 'ejs');

/* ------------- router ----------------------- */
const webRouter = require('./route/web');
app.use(webRouter);

/* ---------------- start server --------------- */
port = process.env.PORT;
app.listen(port, () => {
    //
    console.log("server start at port " + port);
});

the router web.js code :

const express = require('express');
const webRouter = express.Router();

/* ------------- controllers ------------------- */
const staffController = require('../controllers/staffController');
/* -------------- route roles ------------------------ */
webRouter.get('/', (req, res) => {
    res.render('pages/home');
});

webRouter.get('/staff', (req, res) => {
    webRouter.get('/staff', staffController(req, res));
});

// add more route roles

module.exports = webRouter;

staffController.js code :

const staffModel = require('../models/staffModel');

const staffController = (req, res) => {    
    res.render('pages/staff', {
        depName: staffModel().staffTitle,
        staffCount: staffModel().staffCount,
        emps: staffModel().emps
    });
}

module.exports = staffController;
